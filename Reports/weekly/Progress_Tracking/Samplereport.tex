% --- Frame 6: API Contract (1/2) - Request Flow ---
\begin{frame}[fragile]{API Contract: Request Processing}
    \textbf{Step 1: CRM $\rightarrow$ AI Service} \\
    CRM sends candidate metadata and an internal audio URL.
    
    \begin{exampleblock}{POST /api/v1/candidates/analyze}
    \begin{lstlisting}[language=json, basicstyle=\tiny]
{
    "ssn": "079099123456",
    "candidate_name": "Nguyen Van A",
    "audio_url": "http://192.168.1.200/files/interview.mp3"
}
    \end{lstlisting}
    \end{exampleblock}

    \textbf{Step 2: AI Service Response} \\
    The AI acknowledges the request and queues it for background processing.
    
    \begin{block}{Status: 202 Accepted}
    \begin{lstlisting}[language=json, basicstyle=\tiny]
{
    "task_id": "TASK-a1b2c3d4",
    "status": "processing",
    "message": "Audio received; processing in background."
}
    \end{lstlisting}
    \end{block}
\end{frame}

% --- Frame 7: API Contract (2/2) - Webhook Result ---
\begin{frame}[fragile]{API Contract: Webhook Result}
    \textbf{Step 3: AI Service $\rightarrow$ CRM Webhook} \\
    Once analysis is complete, the AI pushes the result back to the CRM.

    \begin{exampleblock}{POST /webhook/candidate-recommendation}
    \begin{lstlisting}[language=json, basicstyle=\tiny]
{
    "task_id": "TASK-a1b2c3d4",
    "ssn": "079099123456",
    "status": "completed",
    "result": {
        "summary": "Solid technical skills. Clear communication.",
        "recommended_message": "HR will follow up within 2 days."
    }
}
    \end{lstlisting}
    \end{exampleblock}

    \begin{itemize}
        \item \scriptsize \textbf{Audio Access:} CRM must host audio on an internal URL reachable by the AI.
        \item \scriptsize \textbf{Reliability:} Recommend exponential-backoff retries for webhook failures.
        \item \scriptsize \textbf{Tracking:} CRM should store the \texttt{task\_id} to match results to candidates.
    \end{itemize}
\end{frame}
